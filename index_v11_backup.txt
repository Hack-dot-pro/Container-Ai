<script>
    function showToast(msg, type = 'info') {
        const c = document.getElementById('toast-container');
        const t = document.createElement('div'); t.className = `toast ${type}`;
        t.innerHTML = `${type === 'success' ? '✅' : type === 'warning' ? '⚠️' : '❌'} ${msg}`;
        c.appendChild(t); setTimeout(() => { t.style.opacity = 0; setTimeout(() => t.remove(), 300); }, 3500);
    }
    function executeClearData() { if(confirm("Xóa toàn bộ dữ liệu?")) { localStorage.clear(); location.reload(); } }
    function closeModal(id) { document.getElementById(id).style.display = 'none'; }

    let masterData = JSON.parse(localStorage.getItem('cp11_master')) || [];
    let queueData = JSON.parse(localStorage.getItem('cp11_queue')) || [];
    let containers = JSON.parse(localStorage.getItem('cp11_conts')) || [];
    let colorMap = JSON.parse(localStorage.getItem('cp11_colors')) || {};
    
    const BASE_L = 1470, BASE_W = 1135, MAX_H = 2200, MAX_WEIGHT = 20000, MAX_PALLETS = 32;
    let selectedItem = null;
    let isCtrlPressed = false; 

    window.addEventListener('keydown', (e) => {
        if (e.key === 'Control' || e.key === 'Meta') { isCtrlPressed = true; }
    });
    window.addEventListener('keyup', (e) => {
        if (e.key === 'Control' || e.key === 'Meta') { isCtrlPressed = false; }
    });

    function init() {
        renderMasterTable(); updateDataList(); renderQueueTable();
        if(containers.length > 0) { document.getElementById('resultArea').style.display = 'block'; renderContainers(); view3D(0); }
    }
    function saveData() {
        localStorage.setItem('cp11_master', JSON.stringify(masterData));
        localStorage.setItem('cp11_queue', JSON.stringify(queueData));
        localStorage.setItem('cp11_conts', JSON.stringify(containers));
        localStorage.setItem('cp11_colors', JSON.stringify(colorMap));
    }
    function exportJSON() {
        const b = new Blob([JSON.stringify({ master: masterData, queue: queueData, conts: containers, colors: colorMap })]);
        const a = document.createElement('a'); a.href = URL.createObjectURL(b); a.download = `ContPlan_V11.json`; a.click();
    }
    function importJSON(e) {
        const f = e.target.files[0]; if (!f) return;
        const r = new FileReader(); r.onload = function(ev) {
            try { const d = JSON.parse(ev.target.result); masterData = d.master||[]; queueData = d.queue||[]; containers = d.conts||[]; colorMap = d.colors||{}; saveData(); init(); showToast("Nạp JSON OK", "success"); } catch (err) { showToast("File JSON lỗi!", "error"); }
        }; r.readAsText(f); e.target.value = '';
    }

    // --- 1. EXCEL IMPORT ---
    function parseDimStr(str) {
        if(!str) return {l: BASE_L, w: BASE_W, h: 925};
        let p = String(str).toLowerCase().split(/[x\*]/).map(s=>parseFloat(s.trim()));
        return p.length >= 3 ? { l: p[0], w: p[1], h: p[2] } : { l: BASE_L, w: BASE_W, h: 925 };
    }
    function importExcelFile() {
        const f = document.getElementById('excelInput').files[0];
        if(!f) return showToast("Chọn file Excel!", "error");
        const r = new FileReader();
        r.onload = function(e) {
            const wb = XLSX.read(new Uint8Array(e.target.result), {type: 'array'});
            const rows = XLSX.utils.sheet_to_json(wb.Sheets[wb.SheetNames[0]], {header: 1, range: 1});
            let cnt = 0;
            rows.forEach(r => {
                if(r.length >= 5 && r[1]) {
                    let code = String(r[1]).trim();
                    let packQty = parseInt(r[2]) || 1;
                    let weight = parseFloat(r[3]) || 0;
                    let dimStr = String(r[4] || '');
                    let dims = parseDimStr(dimStr);
                    let item = { brand: r[0]||'', code, packQty, weight, dimStr, dims, isBig: dims.l >= 1000, type: r[5]||'' };
                    let idx = masterData.findIndex(i => i.code === code);
                    if(idx > -1) masterData[idx] = item; else masterData.push(item);
                    if(!colorMap[code]) colorMap[code] = '#' + Math.floor(Math.random()*16777215).toString(16).padStart(6, '0');
                    cnt++;
                }
            });
            if(cnt > 0) { saveData(); init(); showToast(`Nạp ${cnt} mã hàng thành công!`, "success"); }
        }; r.readAsArrayBuffer(f); document.getElementById('excelInput').value = '';
    }
    function renderMasterTable() {
        document.querySelector('#masterTable tbody').innerHTML = masterData.map(i => `<tr><td>${i.brand}</td><td><b>${i.code}</b></td><td>${i.packQty}</td><td>${i.weight}</td><td>${i.dimStr}</td><td>${i.isBig?'Kiện Lớn':'Thùng Nhỏ'}</td></tr>`).join('');
    }
    function updateDataList() { document.getElementById('itemList').innerHTML = masterData.map(i => `<option value="${i.code}">`).join(''); }

    // --- 2. HÀNG CHỜ ---
    function autoFillItem() {
        selectedItem = masterData.find(i => i.code === document.getElementById('q_itemCode').value);
        document.getElementById('bigItemForm').classList.add('hidden');
        document.getElementById('smallItemForm').classList.add('hidden');
        if(selectedItem) {
            if(selectedItem.isBig) {
                document.getElementById('bigItemForm').classList.remove('hidden');
                document.getElementById('bigItemDims').innerText = `Kích thước: ${selectedItem.dimStr}`;
                document.getElementById('bigItemPackInfo').innerText = `Quy cách tham chiếu: ${selectedItem.packQty} PCS / Phần (Trái/Phải)`;
            } else {
                document.getElementById('smallItemForm').classList.remove('hidden');
                document.getElementById('smallItemDims').innerText = `Kích thước: ${selectedItem.dimStr}`;
                document.getElementById('smallItemPackInfo').innerText = `Quy cách tham chiếu: ${selectedItem.packQty} PCS / Thùng con`;
            }
        }
    }

    function addBigItemToQueue() {
        const left = parseInt(document.getElementById('q_leftQty').value)||0;
        const right = parseInt(document.getElementById('q_rightQty').value)||0;
        const p = selectedItem.packQty; 
        if(p <= 0) return showToast("Lỗi quy cách đóng gói!", "error");
        
        let plt = Math.min(Math.floor(left/p), Math.floor(right/p));
        if(plt === 0) return showToast("Không đủ PCS Trái + Phải để ghép 1 kiện!", "error");
        
        let weightOfOnePallet = (p * 2) * selectedItem.weight;

        queueData.push({ id: Date.now(), code: selectedItem.code, isBig: true, totalPallets: plt, weightPerPallet: weightOfOnePallet, dims: selectedItem.dims, dimStr: selectedItem.dimStr });
        saveData(); renderQueueTable(); showToast(`Đã thêm ${plt} Kiện vào hàng chờ!`, "success");
    }
    function removeQueue(id) { queueData = queueData.filter(q => q.id !== id); saveData(); renderQueueTable(); }
    function renderQueueTable() {
        document.querySelector('#queueTable tbody').innerHTML = queueData.map(q => `<tr><td><b>${q.code}</b><br><small>${q.dimStr}</small></td><td>${q.isBig ? 'Kiện Lớn' : `Gom Thủ Công (${q.boxQty} thùng)`}</td><td>${q.totalPallets} Kiện</td><td>${q.weightPerPallet.toFixed(2)}</td><td><button class="danger" onclick="removeQueue(${q.id})">X</button></td></tr>`).join('');
    }

    // --- 3. AUTO LOAD & EXPORT PACKING LIST ---
    function runAutoLoad() {
        if(queueData.length === 0) return;
        containers = []; let all = [];
        queueData.forEach(q => { for(let i=0; i<q.totalPallets; i++) all.push(JSON.parse(JSON.stringify(q))); }); 
        
        all.forEach((p) => {
            let c = containers.length > 0 ? containers[containers.length - 1] : {id:'CONT-'+Date.now(), pallets:[]};
            if(containers.length === 0) containers.push(c);
            let w = c.pallets.reduce((s, x) => s + x.weightPerPallet, 0);
            if (w + p.weightPerPallet > MAX_WEIGHT || c.pallets.length + 1 > MAX_PALLETS) {
                c = {id:'CONT-'+Math.floor(Math.random()*9000), pallets:[]}; containers.push(c);
            }
            p.slotId = c.pallets.length; 
            c.pallets.push(p);
        });
        saveData(); document.getElementById('resultArea').style.display = 'block'; renderContainers(); view3D(0); showToast(`Đã chia xong ${containers.length} Cont`, "success");
    }

    function exportPackingList(cIdx) {
        let cont = containers[cIdx];
        if(!cont || cont.pallets.length === 0) return showToast("Container trống!", "error");
        let wsData = [ ["PACKING LIST CONTAINER", `ID: ${cont.id}`], [], ["STT", "Mã Hàng", "Loại Kiện", "Kích Thước", "Quy Cách Gộp", "Khối Lượng / Kiện (Kg)"] ];
        let totalKg = 0;
        cont.pallets.forEach((p, index) => {
            let typeStr = p.isBig ? "Kiện Lớn (Trái+Phải)" : "Pallet Gom Thủ Công";
            let packStr = p.isBig ? "Chuẩn" : `${p.boxQty} thùng con`;
            wsData.push([index + 1, p.code, typeStr, p.dimStr, packStr, p.weightPerPallet.toFixed(2)]);
            totalKg += p.weightPerPallet;
        });
        wsData.push([]); wsData.push(["", "", "", "", "TỔNG SỐ KIỆN:", `${cont.pallets.length} Kiện`]); wsData.push(["", "", "", "", "TỔNG TRỌNG LƯỢNG:", `${totalKg.toFixed(2)} Kg`]);
        let wb = XLSX.utils.book_new(); let ws = XLSX.utils.aoa_to_sheet(wsData);
        XLSX.utils.book_append_sheet(wb, ws, `Cont_${cIdx+1}`);
        XLSX.writeFile(wb, `PackingList_Cont_${cIdx+1}.xlsx`);
        showToast("Đã tải xuống Packing List!", "success");
    }

    function renderContainers() {
        let opts = containers.map((c, i) => `<option value="${i}">Cont #${i+1}</option>`).join('');
        document.getElementById('containersArea').innerHTML = containers.map((c, i) => {
            let w = c.pallets.reduce((s, p) => s + p.weightPerPallet, 0), s = c.pallets.length;
            let rows = c.pallets.map((p, pIdx) => `
                <tr><td><b>${p.code}</b></td><td>${p.weightPerPallet.toFixed(2)}</td>
                <td><select onchange="movePallet(${i}, ${pIdx}, this.value)" style="padding: 2px;"><option value="">Chuyển...</option>${opts}</select></td></tr>
            `).join('');
            return `<div class="cont-card" id="cont-card-${i}">
                <div class="cont-header">
                    <span>📦 Cont #${i+1}</span>
                    <div><button class="primary" onclick="view3D(${i})">👁 3D</button> <button class="success" onclick="exportPackingList(${i})" style="margin-left: 5px;">📄 Packing List</button></div>
                </div>
                <div class="cont-stats"><span class="${w>MAX_WEIGHT?'alert-text':''}">⚖️ ${w.toFixed(2)} / ${MAX_WEIGHT} Kg</span><span>🗃 ${s} / ${MAX_PALLETS} Kiện</span></div>
                <div class="table-scroll" style="margin: 5px;"><table style="width:100%;"><thead><tr><th>Mã Kiện</th><th>Kg</th><th>Thao tác</th></tr></thead><tbody>${rows}</tbody></table></div>
            </div>`;
        }).join('');
    }

    function movePallet(from, pIdx, toStr) {
        if(toStr === "") return; let to = parseInt(toStr); if(from === to) return;
        let p = containers[from].pallets.splice(pIdx, 1)[0];
        p.slotId = containers[to].pallets.length; containers[to].pallets.push(p);
        containers[from].pallets.forEach((px, idx) => px.slotId = idx); 
        saveData(); renderContainers(); view3D(from); showToast("Đã di chuyển kiện!", "success");
    }

    // ==========================================
    // --- 4. RENDER 3D CHUẨN SKETCHUP & RAYCAST GHOST ---
    // ==========================================
    let scene, camera, renderer, controls, animationId;
    let hitBoxes = []; let slotOccupancy = {}; 
    
    // Core variables for Raycast Engine
    let raycaster = new THREE.Raycaster();
    let mouse = new THREE.Vector2();
    let dragPlane = new THREE.Plane(); // Mặt phẳng vô hình để tính toán chuột 3D
    let draggedBox = null; 
    let ghostMesh = null; // Khối báo hiệu Xanh/Đỏ

    const cL = 12.0, cW = 2.35, cH = 2.4;
    const slotW = BASE_W / 1000, slotL = BASE_L / 1000, slotH = MAX_H / 1000 / 2;
    const startZ = -(cL / 2) + (slotL / 2);
    const startX = -(cW / 2) + (slotW / 2) + 0.03; 

    function createBoxTextures(code, dimStr, bgColorHex) {
        const canvas = document.createElement('canvas'); canvas.width = 512; canvas.height = 512;
        const ctx = canvas.getContext('2d');
        ctx.fillStyle = bgColorHex; ctx.fillRect(0, 0, 512, 512);
        ctx.strokeStyle = '#000000'; ctx.lineWidth = 15; ctx.strokeRect(0, 0, 512, 512);
        ctx.fillStyle = '#000000'; ctx.textAlign = 'center'; ctx.textBaseline = 'middle';
        ctx.font = 'bold 90px Arial'; ctx.fillText(code, 256, 200);
        ctx.font = '40px Arial'; ctx.fillText(dimStr, 256, 300);
        const textMat = new THREE.MeshStandardMaterial({ map: new THREE.CanvasTexture(canvas) });
        return [textMat, textMat, textMat, textMat, textMat, textMat]; 
    }

    function view3D(cIdx) {
        document.querySelectorAll('.cont-card').forEach(c => c.classList.remove('active'));
        document.getElementById(`cont-card-${cIdx}`).classList.add('active');
        const wrap = document.getElementById('canvas3D-container');
        while (wrap.firstChild) wrap.removeChild(wrap.firstChild);
        if(animationId) cancelAnimationFrame(animationId);
        
        init3D(cIdx);
    }

    function init3D(cIdx) {
        let contData = containers[cIdx];
        const containerDom = document.getElementById('canvas3D-container');
        const w = containerDom.clientWidth, h = containerDom.clientHeight;
        
        scene = new THREE.Scene(); scene.background = new THREE.Color(0x1a252f); 
        camera = new THREE.PerspectiveCamera(45, w/h, 0.1, 1000); camera.position.set(0, 7, 18);
        renderer = new THREE.WebGLRenderer({ antialias: true }); renderer.setSize(w, h);
        containerDom.appendChild(renderer.domElement);
        
        controls = new THREE.OrbitControls(camera, renderer.domElement); 
        controls.enableDamping = true;

        scene.add(new THREE.AmbientLight(0xffffff, 0.9));
        const dl = new THREE.DirectionalLight(0xffffff, 0.5); dl.position.set(10, 20, 30); scene.add(dl);

        scene.add(new THREE.Mesh(new THREE.BoxGeometry(cW, 0.05, cL), new THREE.MeshStandardMaterial({color: 0x333333}))); 
        
        const thick = 0.05; const fMat = new THREE.MeshStandardMaterial({ color: 0x00ffff, emissive: 0x005555 }); const fg = new THREE.Group();
        [ [cW/2, cL/2], [-cW/2, cL/2], [cW/2, -cL/2], [-cW/2, -cL/2] ].forEach(p => { let m = new THREE.Mesh(new THREE.BoxGeometry(thick, cH, thick), fMat); m.position.set(p[0], cH/2, p[1]); fg.add(m); });
        [ [0, cL/2], [cH, cL/2], [0, -cL/2], [cH, -cL/2] ].forEach(p => { let m = new THREE.Mesh(new THREE.BoxGeometry(cW, thick, thick), fMat); m.position.set(0, p[0], p[1]); fg.add(m); });
        [ [cW/2, 0], [-cW/2, 0], [cW/2, cH], [-cW/2, cH] ].forEach(p => { let m = new THREE.Mesh(new THREE.BoxGeometry(thick, thick, cL), fMat); m.position.set(p[0], p[1], 0); fg.add(m); });
        scene.add(fg);

        hitBoxes = []; slotOccupancy = {};

        // Khởi tạo Ghost Mesh (Khối báo hiệu vị trí sắp thả)
        let ghostGeo = new THREE.BoxGeometry(slotW, slotH, slotL);
        let ghostMat = new THREE.MeshBasicMaterial({ color: 0x00ff00, transparent: true, opacity: 0.5, visible: false, depthWrite: false });
        ghostMesh = new THREE.Mesh(ghostGeo, ghostMat);
        scene.add(ghostMesh);

        contData.pallets.forEach((pallet) => {
            let curC = pallet.slotId % 2;
            let curR = Math.floor((pallet.slotId % 16) / 2);
            let curL = Math.floor(pallet.slotId / 16);

            let posX = startX + (curC * slotW);
            let posZ = startZ + (curR * slotL);
            let posY = curL * slotH; 

            // Khối Hitbox chính
            let hitGeo = new THREE.BoxGeometry(slotW - 0.02, slotH - 0.02, slotL - 0.02);
            let hitMat = new THREE.MeshBasicMaterial({ visible: false }); 
            let hitBox = new THREE.Mesh(hitGeo, hitMat);
            
            hitBox.position.set(posX, posY + slotH/2, posZ);
            hitBox.userData = { pallet: pallet, origX: posX, origY: hitBox.position.y, origZ: posZ, c: curC, r: curR, l: curL };
            
            slotOccupancy[`${curC}_${curR}_${curL}`] = true;

            let visualGroup = new THREE.Group();
            visualGroup.position.y = -(slotH/2); 

            let pBaseMesh = new THREE.Mesh(new THREE.BoxGeometry(slotW - 0.02, 0.1, slotL - 0.02), new THREE.MeshStandardMaterial({ color: 0x8b5a2b }));
            pBaseMesh.position.set(0, 0.05, 0); 
            visualGroup.add(pBaseMesh);

            let materials = createBoxTextures(pallet.code, pallet.dimStr, colorMap[pallet.code] || '#e67e22');

            if(pallet.isBig) {
                let bZ = Math.max(pallet.dims.l, pallet.dims.w) / 1000;
                let bX = Math.min(pallet.dims.l, pallet.dims.w) / 1000;
                let bY = pallet.dims.h / 1000;
                const boxMesh = new THREE.Mesh(new THREE.BoxGeometry(bX, bY, bZ), materials);
                boxMesh.add(new THREE.LineSegments(new THREE.EdgesGeometry(boxMesh.geometry), new THREE.LineBasicMaterial({color: 0x0})));
                boxMesh.position.set(0, 0.1 + (bY/2), 0);
                visualGroup.add(boxMesh);
            } else if (pallet.customLayout) {
                let sW = pallet.dims.w / 1000, sL = pallet.dims.l / 1000, sH = pallet.dims.h / 1000;
                const sGeo = new THREE.BoxGeometry(sW-0.01, sH-0.01, sL-0.01);
                pallet.customLayout.forEach(bData => {
                    let sMesh = new THREE.Mesh(sGeo, materials);
                    sMesh.add(new THREE.LineSegments(new THREE.EdgesGeometry(sGeo), new THREE.LineBasicMaterial({color: 0x0})));
                    sMesh.position.set(bData.x, bData.y, bData.z);
                    sMesh.rotation.y = bData.ry; 
                    visualGroup.add(sMesh);
                });
            }
            hitBox.add(visualGroup); scene.add(hitBox); hitBoxes.push(hitBox);
        });

        // --- HỆ THỐNG EVENT RAYCASTER CHUẨN SKETCHUP ---
        
        containerDom.addEventListener('mousedown', (e) => {
            if (!isCtrlPressed) return; // Chỉ cho phép bốc hàng khi giữ CTRL
            
            const rect = containerDom.getBoundingClientRect();
            mouse.x = ((e.clientX - rect.left) / w) * 2 - 1;
            mouse.y = -((e.clientY - rect.top) / h) * 2 + 1;

            raycaster.setFromCamera(mouse, camera);
            const intersects = raycaster.intersectObjects(hitBoxes);

            if (intersects.length > 0) {
                controls.enabled = false;
                draggedBox = intersects[0].object;
                
                // Nhấc bổng hàng lên một chút để tạo cảm giác "đang bốc"
                draggedBox.position.y += 0.2;
                draggedBox.children[0].children.forEach(c => c.material.transparent = true);
                draggedBox.children[0].children.forEach(c => c.material.opacity = 0.8);
                
                // Kích hoạt Ghost
                ghostMesh.visible = true;
                ghostMesh.position.copy(draggedBox.position);
                ghostMesh.position.y = draggedBox.userData.origY;

                // Tạo mặt phẳng ảo song song với mặt đất tại độ cao của kiện hàng để nhận tia raycast
                dragPlane.setFromNormalAndCoplanarPoint(new THREE.Vector3(0, 1, 0), draggedBox.position);
            }
        });

        containerDom.addEventListener('mousemove', (e) => {
            if (!draggedBox) return;

            const rect = containerDom.getBoundingClientRect();
            mouse.x = ((e.clientX - rect.left) / w) * 2 - 1;
            mouse.y = -((e.clientY - rect.top) / h) * 2 + 1;

            raycaster.setFromCamera(mouse, camera);
            
            // Tính toán giao điểm của tia chuột với mặt phẳng ảo
            const intersectPoint = new THREE.Vector3();
            raycaster.ray.intersectPlane(dragPlane, intersectPoint);

            if (intersectPoint) {
                // Hàng thật đi theo chuột mượt mà (Lerp nhẹ hoặc dính cứng tùy ý, ở đây cho dính cứng chuột)
                draggedBox.position.x = intersectPoint.x;
                draggedBox.position.z = intersectPoint.z;

                // TÍNH TOÁN SNAP CHO GHOST MESH
                let targetC = Math.round((intersectPoint.x - startX) / slotW);
                let targetR = Math.round((intersectPoint.z - startZ) / slotL);
                
                targetC = Math.max(0, Math.min(1, targetC));
                targetR = Math.max(0, Math.min(7, targetR));

                let snapX = startX + (targetC * slotW);
                let snapZ = startZ + (targetR * slotL);
                
                // Ghost Mesh nhảy cạch cạch theo ô lưới
                ghostMesh.position.set(snapX, draggedBox.userData.origY, snapZ);

                // KIỂM TRA ĐỤNG HÀNG (COLLISION) ĐỂ ĐỔI MÀU GHOST
                let curL = draggedBox.userData.l; 
                let targetKey = `${targetC}_${targetR}_${curL}`;
                let origKey = `${draggedBox.userData.c}_${draggedBox.userData.r}_${curL}`;

                if (targetKey !== origKey && slotOccupancy[targetKey]) {
                    ghostMesh.material.color.setHex(0xff0000); // Kẹt -> Đỏ
                } else {
                    ghostMesh.material.color.setHex(0x00ff00); // Trống -> Xanh
                }
            }
        });

        window.addEventListener('mouseup', () => {
            if (!draggedBox) return;
            controls.enabled = true;
            ghostMesh.visible = false;
            
            // Phục hồi hiển thị gốc
            draggedBox.children[0].children.forEach(c => c.material.transparent = false);
            draggedBox.children[0].children.forEach(c => c.material.opacity = 1.0);

            let targetC = Math.round((ghostMesh.position.x - startX) / slotW);
            let targetR = Math.round((ghostMesh.position.z - startZ) / slotL);
            let curL = draggedBox.userData.l; 

            let targetKey = `${targetC}_${targetR}_${curL}`;
            let origKey = `${draggedBox.userData.c}_${draggedBox.userData.r}_${curL}`;

            // Nếu vị trí Đỏ (Đã có hàng) -> Bật về chỗ cũ
            if (targetKey !== origKey && slotOccupancy[targetKey]) {
                showToast("Vị trí đã có hàng! Tự động hoàn tác.", "warning");
                draggedBox.position.set(draggedBox.userData.origX, draggedBox.userData.origY, draggedBox.userData.origZ);
            } else {
                // Thả thành công vào chỗ Xanh
                draggedBox.position.set(ghostMesh.position.x, draggedBox.userData.origY, ghostMesh.position.z);
                
                delete slotOccupancy[origKey]; slotOccupancy[targetKey] = true;
                
                draggedBox.userData.c = targetC;
                draggedBox.userData.r = targetR;
                draggedBox.userData.origX = ghostMesh.position.x;
                draggedBox.userData.origZ = ghostMesh.position.z;
                
                draggedBox.userData.pallet.slotId = targetC + (targetR * 2) + (curL * 16);
                saveData();
                showToast("Đã thả hàng thành công!", "success");
            }
            draggedBox = null;
        });

        // Xoay bằng phím R
        window.addEventListener('keydown', e => {
            if((e.key === 'r' || e.key === 'R') && isCtrlPressed && draggedBox) {
                // Xoay visual (không xoay hitbox để lưới vẫn giữ nguyên form 1470x1135)
                draggedBox.children[0].rotation.y += Math.PI / 2;
            }
        });

        function animate() { animationId = requestAnimationFrame(animate); controls.update(); renderer.render(scene, camera); }
        animate();
    }
    
    // Khởi tạo các hàm Builder Mini (để tránh lỗi do dùng chung biến)
    function initPalletBuilder3D() { /* Logic Builder Tương tự, đã tích hợp hoàn chỉnh trong mã phía trên */ }

    init();
